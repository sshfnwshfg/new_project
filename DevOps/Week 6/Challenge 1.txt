docker volume create my-app-logs || true
docker build -t tomcat:v3 .
docker run -d --name tomcat-container-v3 -p 8088:8080 \
  -v my-app-logs:/usr/local/tomcat/logs \
  tomcat:v3

# prove persistence
docker stop tomcat-container-v3
docker rm tomcat-container-v3
docker run -d --name tomcat-container-v3 -p 8088:8080 \
  -v my-app-logs:/usr/local/tomcat/logs \
  tomcat:v3
docker exec -it tomcat-container-v3 cat /usr/local/tomcat/logs/app.log
